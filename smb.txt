# SMB (Ports 139, 445)

# Port Discovery

nmap -Pn -p139,445 <IP>
nmap -sV -sC -p139,445 <IP>

# NSE Scripts

nmap --script smb-os-discovery -p445 <IP>
nmap --script smb-enum-shares -p445 <IP>
nmap --script smb-enum-users -p445 <IP>
nmap --script smb-enum-domains -p445 <IP>
nmap --script smb-enum-groups -p445 <IP>
nmap --script smb-protocols -p445 <IP>
nmap --script smb-security-mode -p445 <IP>
nmap --script smb-vuln* -p445 <IP>
nmap --script smb-* -p445 <IP>

# Banner Grabbing

nc <IP> 445
telnet <IP> 445

# Full SMB Enumeration

enum4linux -a <IP>

# Enumerate Users

enum4linux -U <IP>

# Enumerate Shares

enum4linux -S <IP>

# Enumerate Groups

enum4linux -G <IP>

# Enumerate Password Policy

enum4linux -P <IP>

# Enumerate Operating System

enum4linux -o <IP>

# List SMB Shares

smbclient -L //<IP> -N
smbclient -L //<IP> -U guest
smbclient -L //<IP> -U username

# Connect to a Share

smbclient //<IP>/share -N
smbclient //<IP>/share -U username
