# Port Discovery
nmap -Pn -p5671 <IP>
nmap -sV -sC -p5671 <IP>

# AMQPS / TLS Identification
nmap --script amqp-info -p5671 <IP>
nmap --script banner -p5671 <IP>
nmap --script ssl-cert -p5671 <IP>
nmap --script ssl-enum-ciphers -p5671 <IP>
nmap --script ssl-* -p5671 <IP>

# TLS / SSL Testing
openssl s_client -connect <IP>:5671
openssl s_client -connect <IP>:5671 -servername <IP>
sslscan <IP>:5671
testssl.sh <IP>:5671

# Certificate Inspection
openssl s_client -connect <IP>:5671 -showcerts
openssl s_client -connect <IP>:5671 2>/dev/null | openssl x509 -noout -text
openssl s_client -connect <IP>:5671 2>/dev/null | openssl x509 -noout -issuer -subject -dates

# Cipher / Protocol Testing
nmap --script ssl-enum-ciphers -p5671 <IP>
testssl.sh --protocols <IP>:5671
testssl.sh --ciphers <IP>:5671
testssl.sh --certificate <IP>:5671
